import json

with open('DATABASE_RECIPIENTS.json', 'r') as f:
    db = json.load(f)

# Get all active recipients for today
admin_recipients = db['admin_recipients']
customer_recipients = db['customer_recipients']

# Filter by status and frequency
active_emails = [
    r['email'] for r in (admin_recipients + customer_recipients)
    if r['status'] == 'active' and r['frequency'] == 'daily'
]

# Send to all active emails
for email in active_emails:
    send_report_email(email, report_data)
